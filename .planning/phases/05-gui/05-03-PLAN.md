- Member: float currentValue (0-1), juce::String sourceLabel
- paint():

  - Source label on the left side of the bar
  - Percentage text on the right side ("XX%"), always visible

- mouseDown(): attachment.beginGesture()

- mouseUp(): attachment.endGesture()
